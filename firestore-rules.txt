rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /chamados/{chamadoId} {
      allow create: if request.resource.data.keys().hasAll([
        'protocolo', 'nomeSolicitante', 'setor', 'tipo', 'prioridade',
        'assunto', 'descricao', 'status', 'criadoEm'
      ])
      && request.resource.data.status == 'Aberto'
      && request.resource.data.nomeSolicitante is string
      && request.resource.data.setor is string
      && request.resource.data.assunto is string
      && request.resource.data.descricao is string;
      allow read, update, delete: if request.auth != null;
    }
    match /chamados/{chamadoId}/mensagens/{mensagemId} {
      allow read, write: if request.auth != null;
    }
  }
}
